        if(NOT EXISTS "${NAVIDOG_MYSQL_CLIENT_ROOT}/lib/pkgconfig/mysqlclient.pc")
            message(FATAL_ERROR "NAVIDOG_MYSQL_CLIENT_ROOT must contain lib/pkgconfig/mysqlclient.pc")
        endif()
        set(ENV{PKG_CONFIG_PATH} "${NAVIDOG_MYSQL_CLIENT_ROOT}/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
    endif()
    if(NOT "${NAVIDOG_LAST_MYSQL_CLIENT_ROOT}" STREQUAL "${NAVIDOG_MYSQL_CLIENT_ROOT}")
        unset(__pkg_config_checked_MYSQL CACHE)
        unset(pkgcfg_lib_MYSQL_mysqlclient CACHE)
    endif()
    pkg_check_modules(MYSQL QUIET IMPORTED_TARGET mysqlclient)
    set(ENV{PKG_CONFIG_PATH} "${navidog_saved_pkg_path}")
    set(NAVIDOG_LAST_MYSQL_CLIENT_ROOT "${NAVIDOG_MYSQL_CLIENT_ROOT}" CACHE INTERNAL "Selected native MySQL SDK" FORCE)
    if(MYSQL_FOUND)
        # Some distributions expose MariaDB Connector/C as mysqlclient. The
        # query adapter requires the Oracle client's nonblocking API.
        include(CheckCXXSourceCompiles)
        include(CMakePushCheckState)
        cmake_push_check_state(RESET)
        set(CMAKE_REQUIRED_LIBRARIES PkgConfig::MYSQL)
        foreach(check NAVIDOG_MYSQL_ASYNC_API NAVIDOG_MYSQL_HAS_SNI)
            unset(${check} CACHE)
        endforeach()
        check_cxx_source_compiles([=[
            #include <mysql.h>
            int main() {
                MYSQL *mysql = nullptr;
                return mysql_next_result_nonblocking(mysql);
            }
        ]=] NAVIDOG_MYSQL_ASYNC_API)
        if(NAVIDOG_MYSQL_ASYNC_API)
            check_cxx_source_compiles([=[
                #include <mysql.h>
                int main() { return MYSQL_OPT_TLS_SNI_SERVERNAME; }
            ]=] NAVIDOG_MYSQL_HAS_SNI)
        else()
            set(MYSQL_FOUND FALSE)
            message(STATUS "Navidog: skipping mysqlclient without the MySQL nonblocking API")
        endif()
        cmake_pop_check_state()
    endif()
    if(MYSQL_FOUND)
        target_link_libraries(navidog_core PRIVATE PkgConfig::MYSQL)
        target_compile_definitions(navidog_core PRIVATE NAVIDOG_HAS_MYSQL=1)
        if(NAVIDOG_MYSQL_HAS_SNI)
            target_compile_definitions(navidog_core PRIVATE NAVIDOG_MYSQL_HAS_SNI=1)
        endif()
        find_package(OpenSSL REQUIRED)
        target_link_libraries(navidog_core PRIVATE OpenSSL::SSL OpenSSL::Crypto)
        target_include_directories(navidog_core BEFORE PRIVATE ${OPENSSL_INCLUDE_DIR})
    endif()
endif()
