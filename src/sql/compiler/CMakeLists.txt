# Can also be built without Qt/drivers through NAVIDOG_COMPILER_ONLY.
add_library(navidog_sql_frontend STATIC
    temporal.cpp types.cpp type_inference.cpp inspection.cpp transaction.cpp prepared.cpp json_table.cpp operator_catalog.cpp merge.cpp script_context.cpp semantic.cpp grouping.cpp query_features.cpp completion.cpp conversion.cpp outline.cpp routine.cpp routine_printer.cpp
    format_style.cpp formatter.cpp context.cpp lexer.cpp parser.cpp dml.cpp ddl.cpp joins.cpp query_extensions.cpp model.cpp model_ddl.cpp dump.cpp dump_ddl.cpp printer.cpp
    frontends/registry.cpp frontends/mysql.cpp frontends/postgresql.cpp
    frontends/oracle.cpp frontends/sqlite.cpp frontends/sqlserver.cpp
    frontends/mariadb.cpp frontends/oceanbase.cpp frontends/tidb.cpp)
target_include_directories(navidog_sql_frontend PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../../include")
target_compile_features(navidog_sql_frontend PUBLIC cxx_std_17)
set_target_properties(navidog_sql_frontend PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF AUTOMOC OFF)
if(MSVC)
    target_compile_options(navidog_sql_frontend PRIVATE /W4 /utf-8)
else()
    target_compile_options(navidog_sql_frontend PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(navidog-sql "${CMAKE_CURRENT_SOURCE_DIR}/../../../apps/sql/main.cpp")
target_link_libraries(navidog-sql PRIVATE navidog_sql_optimizer)
set_target_properties(navidog-sql PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF
    AUTOMOC OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
